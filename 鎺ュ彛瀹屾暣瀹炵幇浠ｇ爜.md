## 0. 先做 3 个准备改动（不改会导致编译失败或返回结构不符）

### 0.1 增强统一响应 `R`（替换 `src/main/java/com/ar/contractreview/result/R.java`）

把 `data` 从 `Map` 改成 `Object`，加 `data(Object)` 重载，并补 `timestamp`：

```java
package com.ar.contractreview.result;

import lombok.Data;
import java.util.HashMap;
import java.util.Map;

/**
 * 统一响应结果封装类
 */
@Data
public class R {

    private Integer code;
    private String message;
    // data: 可以是对象、数组、Map、分页结果
    private Object data;
    // 时间戳（与接口文档一致）
    private Long timestamp = System.currentTimeMillis();

    private R() {}

    public static R ok() {
        R r = new R();
        r.setCode(ResponseCode.SUCCESS.getCode());
        r.setMessage(ResponseCode.SUCCESS.getMessage());
        return r;
    }

    /** 把对象/数组/分页对象直接放到 data 根部 */
    public R data(Object obj) {
        this.setCode(ResponseCode.SUCCESS.getCode());
        this.setMessage(ResponseCode.SUCCESS.getMessage());
        this.data = obj;
        return this;
    }

    /** 键值对：data={"key":value} */
    public R data(String key, Object value) {
        this.setCode(ResponseCode.SUCCESS.getCode());
        this.setMessage(ResponseCode.SUCCESS.getMessage());
        if (data == null || !(data instanceof Map)) data = new HashMap<>();
        ((Map<String, Object>) data).put(key, value);
        return this;
    }

    /** 批量键值对 */
    public R data(Map<String, Object> dataMap) {
        this.setCode(ResponseCode.SUCCESS.getCode());
        this.setMessage(ResponseCode.SUCCESS.getMessage());
        if (data == null || !(data instanceof Map)) data = new HashMap<>();
        ((Map<String, Object>) data).putAll(dataMap);
        return this;
    }

    public static R fail() {
        R r = new R();
        r.setCode(ResponseCode.FAIL.getCode());
        r.setMessage(ResponseCode.FAIL.getMessage());
        return r;
    }
    public static R fail(Integer code, String message) {
        R r = new R();
        r.setCode(code); r.setMessage(message);
        return r;
    }
    public static R fail(ResponseCode responseCode) {
        R r = new R();
        r.setCode(responseCode.getCode());
        r.setMessage(responseCode.getMessage());
        return r;
    }
}
```

> 若**不改 `R`**：把本文代码里所有 `.data(obj)` 替换为 `.data(JSON.parseObject(JSON.toJSONString(obj), Map.class))`，把 `.data(list)` 替换为 `.data("list", list)`。为了简洁，下文统一按「已改 `R`」编写。

```java
        user.setId(id);
```

```java
@Service
public class SysMenuServiceImpl extends ServiceImpl<SysMenuMapper, SysMenu> implements SysMenuService {

    @Autowired private SysRoleMenuMapper roleMenuMapper;

    @Override public List<SysMenu> listMenu() {
        return this.list(new LambdaQueryWrapper<SysMenu>().orderByAsc(SysMenu::getSortOrder));
    }
    @Override public List<Map<String,Object>> tree() {
        List<SysMenu> all = listMenu();
        Map<Long, Map<String,Object>> nodeMap = new HashMap<>();
        List<Map<String,Object>> roots = new ArrayList<>();
        for (SysMenu m : all) {
            Map<String,Object> n = new HashMap<>();
            n.put("id", m.getId()); n.put("parentId", m.getParentId());
            n.put("menuName", m.getMenuName()); n.put("menuType", m.getMenuType());
            n.put("path", m.getPath()); n.put("icon", m.getIcon());
            n.put("children", new ArrayList<>());
            nodeMap.put(m.getId(), n);
        }
        for (SysMenu m : all) {
            Map<String,Object> n = nodeMap.get(m.getId());
            if (m.getParentId() != null && m.getParentId() != 0 && nodeMap.containsKey(m.getParentId())) {
                ((List<Object>) nodeMap.get(m.getParentId()).get("children")).add(n);
            } else {
                roots.add(n);
            }
        }
        return roots;
    }
    @Override public List<SysMenu> listByRole(Long roleId) {
        List<Long> menuIds = roleMenuMapper.selectList(new LambdaQueryWrapper<SysRoleMenu>().eq(SysRoleMenu::getRoleId, roleId))
                .stream().map(SysRoleMenu::getMenuId).collect(Collectors.toList());
        return menuIds.isEmpty() ? new ArrayList<>() : this.listByIds(menuIds);
    }
}
```

```java
@Service
public class ContractContractServiceImpl extends ServiceImpl<ContractContractMapper, ContractContract>
        implements ContractContractService {

    @Override public IPage<ContractContract> pageContract(Integer page, Integer size, String contractNo, String contractName,
            String contractType, String status, String riskLevel, Long uploadUserId, String startTime, String endTime) {
        LambdaQueryWrapper<ContractContract> qw = new LambdaQueryWrapper<>();
        qw.like(StringUtils.hasText(contractNo), ContractContract::getContractNo, contractNo)
          .like(StringUtils.hasText(contractName), ContractContract::getContractName, contractName)
          .eq(StringUtils.hasText(contractType), ContractContract::getContractType, contractType)
          .eq(StringUtils.hasText(status), ContractContract::getStatus, status)
          .eq(StringUtils.hasText(riskLevel), ContractContract::getRiskLevel, riskLevel)
          .eq(uploadUserId != null, ContractContract::getUploadUserId, uploadUserId)
          .apply(StringUtils.hasText(startTime), "upload_time >= {0}", startTime)
          .apply(StringUtils.hasText(endTime), "upload_time <= {0}", endTime)
          .orderByDesc(ContractContract::getUploadTime);
        return this.page(new Page<>(page, size), qw);
    }
    @Override public ContractContract submitReview(Long id, Long reviewerId, String comment) {
        ContractContract c = this.getById(id);
        if (c == null) throw new BusinessException(ResponseCode.PARAMETER_EXCEPTION);
        c.setStatus("REVIEWING").setReviewerId(reviewerId).setReviewComment(comment);
        this.updateById(c);
        return c;
    }
    @Override public ContractContract approve(Long id, String comment) {
        ContractContract c = this.getById(id);
        if (c == null) throw new BusinessException(ResponseCode.PARAMETER_EXCEPTION);
        c.setStatus("APPROVED").setReviewComment(comment).setReviewTime(LocalDateTime.now());
        this.updateById(c);
        return c;
    }
    @Override public ContractContract reject(Long id, String comment) {
        ContractContract c = this.getById(id);
        if (c == null) throw new BusinessException(ResponseCode.PARAMETER_EXCEPTION);
        c.setStatus("REJECTED").setReviewComment(comment).setReviewTime(LocalDateTime.now());
        this.updateById(c);
        return c;
    }
```

```java
                    @RequestParam String partyA, @RequestParam String partyB,
                    @RequestParam BigDecimal amount,
                    @RequestParam(required = false) String signDate,
                    @RequestParam(required = false) String startDate,
```
